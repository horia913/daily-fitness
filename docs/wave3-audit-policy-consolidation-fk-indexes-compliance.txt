  cmd: UPDATE | roles: {public}

PROPOSED MINIMAL SET (≤1 permissive per cmd for authenticated/public as today):

-- SELECT: wse_select_authenticated (roles: authenticated, public — match current app JWT usage)
CREATE POLICY wse_select_authenticated ON workout_set_entries
  FOR SELECT TO public
  USING (
    EXISTS (
      SELECT 1 FROM workout_templates wt
      WHERE wt.id = workout_set_entries.template_id
        AND wt.coach_id = (SELECT auth.uid())
    )
    OR EXISTS (
      SELECT 1 FROM workout_assignments wa
      WHERE wa.workout_template_id = workout_set_entries.template_id
        AND wa.client_id = (SELECT auth.uid())
    )
    OR EXISTS (
      SELECT 1
      FROM program_assignments pa
      JOIN program_schedule ps ON ps.program_id = pa.program_id
      WHERE pa.client_id = (SELECT auth.uid())
        AND pa.status = 'active'
        AND ps.template_id = workout_set_entries.template_id
    )
    OR (SELECT is_admin((SELECT auth.uid())))
  );

-- WRITE (coach template owner): single ALL policy OR keep INSERT/UPDATE/DELETE with identical expr
CREATE POLICY wse_coach_manage ON workout_set_entries
  FOR ALL TO public
  USING (
    EXISTS (
      SELECT 1 FROM workout_templates wt
      WHERE wt.id = workout_set_entries.template_id
        AND wt.coach_id = (SELECT auth.uid())
    )
  )
  WITH CHECK (
    EXISTS (
      SELECT 1 FROM workout_templates wt
      WHERE wt.id = workout_set_entries.template_id
        AND wt.coach_id = (SELECT auth.uid())
    )
  );
